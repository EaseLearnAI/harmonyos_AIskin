### 二、动态权限申请
在Ability的`onWindowStageCreate`阶段申请权限：
```typescript
import abilityAccessCtrl from '@kit.AbilityKit';

async function requestCameraPermission(context: common.UIAbilityContext) {
  const atManager = abilityAccessCtrl.createAtManager();
  try {
    const granted = await atManager.requestPermissionsFromUser(context, [
      'ohos.permission.CAMERA',
      'ohos.permission.WRITE_USER_STORAGE'
    ]);
    return granted === 0; // PERMISSION_GRANTED为0
  } catch (err) {
    console.error('权限申请失败:', err);
    return false;
  }
}
```

**三、摄像头调用与拍照**
实现摄像头预览及拍照功能：
```typescript
import camera from '@kit.CameraKit';
import image from '@kit.ImageKit';

class CameraHandler {
  private cameraManager: camera.CameraManager;
  private cameraSession: camera.CameraSession;

  // 初始化摄像头
  async initCamera(previewSurfaceId: string) {
    this.cameraManager = camera.getCameraManager();
    const cameras = this.cameraManager.getSupportedCameras();
    const cameraInput = this.cameraManager.createCameraInput(cameras);

    this.cameraSession = this.cameraManager.createCaptureSession();
    const previewOutput = this.cameraManager.createPreviewOutput(previewSurfaceId);
    
    await this.cameraSession.beginConfig();
    this.cameraSession.addInput(cameraInput);
    this.cameraSession.addOutput(previewOutput);
    await this.cameraSession.commitConfig();
    await this.cameraSession.start();
  }

  // 拍照并保存
  async takePhoto(): Promise<string> {
    const context = getContext() as common.UIAbilityContext;
    const photoPath = `${context.filesDir}/photo_${Date.now()}.jpg`;
    
    const photoOutput = this.cameraManager.createPhotoOutput({
      rotation: image.ImageRotation.ROTATION_0,
      quality: image.QualityLevel.QUALITY_LEVEL_HIGH,
      location: null,
      mirror: false
    }, photoPath);

    await this.cameraSession.addOutput(photoOutput);
    await photoOutput.capture();
    return photoPath;
  }
}
```

**四、图片上传实现**
使用HTTP模块上传文件：
```typescript
import http from '@kit.NetworkKit';
import fileIo from '@ohos.fileio';

async function uploadImage(filePath: string) {
  const file = fileIo.openSync(filePath, fileIo.OpenMode.READ_ONLY);
  const buffer = new ArrayBuffer(file.stat().size);
  fileIo.readSync(file.fd, buffer);

  const uploadOptions = {
    method: http.RequestMethod.POST,
    header: { 'Content-Type': 'multipart/form-data' },
    extraData: { 
      file: { 
         buffer, 
        filename: 'captured_image.jpg',
        contentType: 'image/jpeg' 
      }
    }
  };

  const response = await http.request('https://your-api.com/upload', uploadOptions);
  return response.result;
}
```

**五、UI整合示例**
在ArkUI中绑定预览组件：
```typescript
@Entry
@Component
struct CameraPage {
  private xComponentController: XComponentController = new XComponentController();
  private cameraHandler: CameraHandler = new CameraHandler();

  aboutToAppear() {
    // 申请权限后初始化摄像头
    requestCameraPermission(getContext()).then((granted) => {
      if (granted) {
        this.cameraHandler.initCamera(this.xComponentController.getXComponentSurfaceId());
      }
    });
  }

  build() {
    Column() {
      // 预览区域
      XComponent({
        id: 'cameraPreview',
        controller: this.xComponentController,
        type: 'surface',
        libraryname: 'camera'
      }).width('100%').height('60%')

      // 拍照按钮
      Button('拍照上传')
        .onClick(async () => {
          const photoPath = await this.cameraHandler.takePhoto();
          await uploadImage(photoPath);
        })
    }
  }
}
```
